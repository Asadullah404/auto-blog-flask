Google Apps Script — 30-Second Setup Guide
⚡ Simple 3-Step Setup (Zero Google Cloud Console Needed):
  1. Open your Google Sheet and click Extensions > Apps Script.
  2. Delete any default code, paste the script below, and click Save (Ctrl+S).
  3. Click Deploy > New deployment, select type Web app, configure:
    • Execute as: Me
    • Who has access: Anyone
    Then click Deploy and paste the resulting Web App URL into Settings!
Apps Script Gateway Code: 
/**
 * Auto Content Pipeline — Google Sheets Sync Gateway
 * Paste this into Extensions > Apps Script in your Google Sheet, then Deploy as Web App.
 * IMPORTANT: When deploying, set "Execute as: Me" and "Who has access: Anyone".
 */
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    message: "Google Sheets Sync Gateway is active and ready! Use POST to sync."
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: "Sheet is busy with another sync request. Please try again."
    })).setMimeType(ContentService.MimeType.JSON);
  }
  
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var payload = JSON.parse(e.postData.contents);
    
    // 1. Healthcheck / Ping
    if (payload.action === "ping") {
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        message: "Connected to sheet: " + SpreadsheetApp.getActiveSpreadsheet().getName()
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    var rows = payload.rows || [];
    var headers = ["ID", "Source URL", "Category", "Status", "Published Link", "Created At", "Updated At"];
    var lastRow = sheet.getLastRow();
    
    // 2. Initialize Headers if sheet is blank
    if (lastRow === 0) {
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length)
           .setFontWeight("bold")
           .setBackground("#1e293b")
           .setFontColor("#f8fafc");
      sheet.setFrozenRows(1);
      lastRow = 1;
    }
    
    // 3. Build Index of Existing URLs in Column B (Index 2 in sheet)
    var urlMap = {};
    if (lastRow > 1) {
      var existingUrls = sheet.getRange(2, 2, lastRow - 1, 1).getValues();
      for (var i = 0; i < existingUrls.length; i++) {
        var u = String(existingUrls[i][0] || "").trim();
        if (u) {
          urlMap[u] = i + 2; // 1-based row index in sheet
        }
      }
    }
    
    var updatedCount = 0;
    var addedCount = 0;
    var toAppend = [];
    
    // 4. Match & Synchronize
    for (var j = 0; j < rows.length; j++) {
      var r = rows[j];
      var srcUrl = String(r.source_url || r.url || "").trim();
      if (!srcUrl) continue;
      
      var rowData = [
        r.id || "",
        srcUrl,
        r.category || "General",
        r.status || "pending",
        r.published_link || r.post_url || "N/A",
        r.created_at || "",
        r.updated_at || ""
      ];
      
      if (urlMap[srcUrl]) {
        // Update in-place to avoid duplicate rows
        var targetRow = urlMap[srcUrl];
        sheet.getRange(targetRow, 1, 1, rowData.length).setValues([rowData]);
        updatedCount++;
      } else {
        toAppend.push(rowData);
        urlMap[srcUrl] = lastRow + toAppend.length; // track to prevent duplicates within batch
        addedCount++;
      }
    }
    
    // 5. Append new rows in a single batch
    if (toAppend.length > 0) {
      var nextRow = sheet.getLastRow() + 1;
      sheet.getRange(nextRow, 1, toAppend.length, headers.length).setValues(toAppend);
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      total: rows.length,
      updated: updatedCount,
      added: addedCount,
      sheet_name: SpreadsheetApp.getActiveSpreadsheet().getName()
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
